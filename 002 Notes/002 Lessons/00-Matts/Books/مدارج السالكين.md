###### الإقتباسات
```dataviewjs
// 1. الحصول على الملاحظة الحالية ومسارها
const currentNote = dv.current();
const currentPath = currentNote.file.path;
const filepath = `"002 Notes/002 Lessons/00-Matts/Quote's"`;

// 2. جلب جميع الملفات من المجلد المطلوب
const files = dv.pages(filepath);

// 3. مصفوفة لتخزين النتائج (كل صف في الجدول)
let rows = [];

// 4. التكرار على كل ملف
for (let file of files) {
    // التحقق من وجود خاصية book
    if (!file.book) continue;
    
    // استخراج مسار الكتاب من خاصية book (قد تكون كائن link أو مصفوفة)
    let bookPath = null;
    if (typeof file.book === 'object' && file.book.path) {
        bookPath = file.book.path;
    } else if (Array.isArray(file.book) && file.book[0]?.path) {
        bookPath = file.book[0].path;
    } else if (typeof file.book === 'string') {
        // في حالة كانت نصاً عادياً (كاحتمال احتياطي)
        bookPath = file.book.replace(/\[\[|\]\]/g, '');
    }
    
    // مقارنة المسار مع مسار الملاحظة الحالية
    if (bookPath !== currentPath) continue;
    
    // قراءة المحتوى الخام للملف
    const rawContent = await dv.io.load(file.file.path);
    if (!rawContent) continue;
    
    // تقسيم المحتوى إلى أسطر واستخراج الاقتباسات فقط
    const lines = rawContent.split('\n');
    for (let line of lines) {
        const trimmedLine = line.trim();
        // إذا بدأ السطر بـ > فهو اقتباس
        if (trimmedLine.startsWith('>') && !trimmedLine.startsWith('> [!bookquote]')) {
            // إزالة علامات > من البداية (واحدة أو أكثر مع المسافات)
            let quoteText = trimmedLine.replace(/^>+\s*/, '');
            
            if (quoteText.length > 0) {
                
                // === التعديل الجديد: التحكم في طول الاقتباس ===
                const maxLength = 200; // (يمكنك تعديل هذا الرقم ليتناسب تماماً مع سطرين في شاشتك)
                
                if (quoteText.length > maxLength) {
                    // قص النص عند الحد الأقصى
                    let truncated = quoteText.substring(0, maxLength);
                    
                    // البحث عن آخر مسافة لتجنب قطع الكلمات في المنتصف
                    const lastSpaceIndex = truncated.lastIndexOf(" ");
                    if (lastSpaceIndex > 0) {
                        truncated = truncated.substring(0, lastSpaceIndex);
                    }
                    
                    // إضافة النقاط
                    quoteText = truncated + "...";
                }
                // ==============================================

                rows.push([file.file.link, quoteText]);
            }
        }
    }
}

// 5. عرض النتائج في جدول
if (rows.length > 0) {
    dv.table(["الملف", "الاقتباس"], rows);
} else {
    dv.paragraph(`❌ لا توجد اقتباسات في الملفات المرتبطة بـ "[[${currentNote.file.name}]]".`);
}
```
