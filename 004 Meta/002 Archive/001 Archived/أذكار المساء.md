> [!adhkar] آية الكرسي
> **أَعُوذُ بِاللَّهِ مِنَ الشَّيطَانِ الرَّجِيمِ** ﴿اللَّهُ لاَ إِلَهَ إِلاَّ هُوَ الْحَيُّ الْقَيُّومُ لاَ تَأْخُذُهُ سِنَةٌ وَلاَ نَوْمٌ لَّهُ مَا فِي السَّمَوَاتِ وَمَا فِي الأَرْضِ مَن ذَا الَّذِي يَشْفَعُ عِنْدَهُ إِلاَّ بِإِذْنِهِ يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلاَ يُحِيطُونَ بِشَيْءٍ مِّنْ عِلْمِهِ إِلاَّ بِمَا شَاء وَسِعَ كُرْسِيُّهُ السَّمَوَاتِ وَالْأَرْضَ وَلاَ يَؤُودُهُ حِفْظُهُمَا وَهُوَ الْعَلِيُّ الْعَظِيمُ﴾
> *(مرة واحدة)*

> [!adhkar] سورة الإخلاص والمعوذات
> **بسم الله الرحمن الرحيم**  ﴿قُلْ هُوَ اللَّهُ أَحَدٌ اللَّهُ الصَّمَدُ لَمْ يَلِدْ وَلَمْ يُولَدْ وَلَمْ يَكُن لَّهُ كُفُواً أَحَدٌ﴾.
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] سورة الفلق 
> **بسم الله الرحمن الرحيم** ﴿قُلْ أَعُوذُ بِرَبِّ الْفَلَقِ مِن شَرِّ مَا خَلَقَ وَمِن شَرِّ غَاسِقٍ إِذَا وَقَبَ وَمِن شَرِّ النَّفَّاثَاتِ فِي الْعُقَدِ وَمِن شَرِّ حَاسِدٍ إِذَا حَسَدَ﴾.
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] سورة الناس
> **بسم الله الرحمن الرحيم** ﴿قُلْ أَعُوذُ بِرَبِّ النَّاسِ مَلِكِ النَّاسِ إِلَهِ النَّاسِ مِن شَرِّ الْوسْوَاسِ الْخَنَّاسِ الَّذِي يُوَسْوِسُ فِي صُدُورِ النَّاسِ مِنَ الْجِنَّةِ وَ النَّاسِ﴾
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] ٱلْحَمْدُ لِلَّهِ
> ٱلْحَمْدُ لِلَّهِ وَحْدَهُ، وَٱلصَّلَاةُ وَٱلسَّلَامُ عَلَىٰ مَنْ لَا نَبِيَّ بَعْدَهُ
> *(مرة واحدة)*

> [!adhkar] أمسينا وأمسى الملك لله  
> ((أَمْسَيْنَا وَأَمْسَى الْمُلْكُ لِلَّهِ ، وَالْحَمْدُ لِلَّهِ، لاَ إِلَهَ إلاَّ اللَّهُ وَحْدَهُ لاَ شَرِيكَ لَهُ، لَهُ الْمُلْكُ وَلَهُ الْحَمْدُ وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌ، رَبِّ أَسْأَلُكَ خَيْرَ مَا فِي هَذِهِ اللَّيْلَةِ وَخَيْرَ مَا بَعْدَهَا ، وَأَعُوذُ بِكَ مِنْ شَرِّ مَا فِي هَذِهِ اللَّيْلَةِ وَشَرِّ مَا بَعْدَهَا، رَبِّ أَعُوذُ بِكَ مِنَ الْكَسَلِ وَسُوءِ الْكِبَرِ، رَبِّ أَعُوذُ بِكَ مِنْ عَذَابٍ فِي النَّارِ وَعَذَابٍ فِي الْقَبْرِ)).  
> *(مرة واحدة)*

> [!adhkar] أمسينا بك وأصبحنا بك  
> ((اللَّهُمَّ بِكَ أَمْسَيْنَا، وَبِكَ أَصْبَحْنَا ، وَبِكَ نَحْيَا، وَبِكَ نَمُوتُ وَإِلَيْكَ الْمَصِيرُ)).  
> *(مرة واحدة)*

> [!adhkar] سيد الاستغفار  
> ((اللَّهُمَّ أَنْتَ رَبِّي لَا إِلَهَ إِلاَّ أَنْتَ، خَلَقْتَنِي وَأَنَا عَبْدُكَ، وَأَنَا عَلَى عَهْدِكَ وَوَعْدِكَ مَا اسْتَطَعْتُ، أَعُوذُ بِكَ مِنْ شَرِّ مَا صَنَعْتُ، أَبُوءُ لَكَ بِنِعْمَتِكَ عَلَيَّ، وَأَبُوءُ بِذَنْبِي فَاغْفِرْ لِي فَإِنَّهُ لاَ يَغْفِرُ الذُّنوبَ إِلاَّ أَنْتَ)).  
> *(مرة واحدة)*

> [!adhkar] دعاء الكرب  
> ((اللَّهُمَّ أَنْتَ رَبِّي لاَ إِلَهَ إِلاَّ أَنْتَ، عَلَيْكَ تَوَكَّلْتُ، وَأَنْتَ رَبُّ الْعَرْشِ الْعَظِيمِ، مَا شَاءَ اللَّهُ كَانَ، وَمَا لَمْ يَشَأْ لَمْ يَكُنْ، لاَ حَوْلَ وَلاَ قُوَّةَ إِلاَّ بِاللَّهِ، أَعْلَمُ أَنَّ اللَّهَ عَلَى كُلِّ شَيْءٍ قَدِيرٌ، وَأَنَّ اللَّهَ قَدْ أَحَاطَ بِكُلِّ شَيْءٍ عِلْمًا، اللَّهُمَّ إِنِّي أَعُوذُ بِكَ مِنْ شَرِّ نَفْسِي، وَمِنْ شَرِّ كُلِّ دَابَّةٍ أَنْتَ آخِذٌ بِنَاصِيَتِهَا، إِنَّ رَبِّي عَلَى صِرَاطٍ مُسْتَقِيمٍ)).  
> *(مرة واحدة)*

> [!adhkar] سؤال خير الليلة  
> ((أَمْسَيْنَا وَأَمْسَى الْمُلْكُ لِلَّهِ رَبِّ الْعَالَمِينَ، اللَّهُـمَّ إِنِّي أَسْأَلُكَ خَيْرَ هَذِهِ اللَّيْلَةِ: فَتْحَهَا، وَنَصْرَهَا، وَنُورَهَا، وَبَرَكَتَهَا، وَهُدَاهَا، وَأَعُوذُ بِكَ مِنْ شَرِّ مَا فِيهَا وَشَرِّ مَا بَعْدَهَا)).  
> *(مرة واحدة)*

> [!adhkar] الفطرة والإخلاص  
> ((أَمْسَيْنَا عَلَى فِطْرَةِ الْإِسْلاَمِ، وَعَلَى كَلِمَةِ الْإِخْلاَصِ، وَعَلَى دِينِ نَبِيِّنَا مُحَمَّدٍ ﷺ، وَعَلَى مِلَّةِ أَبِينَا إِبْرَاهِيمَ، حَنِيفاً مُسْلِماً وَمَا كَانَ مِنَ الْمُشرِكِينَ))
>*(مرة واحدة)*

> [!adhkar] الشكر على النعمة  
> ((اللَّهُمَّ مَا أَمْسَى بِي مِنْ نِعْمَةٍ أَوْ بِأَحَدٍ مِنْ خَلْقِكَ فَمِنْكَ وَحْدَكَ لاَ شَرِيكَ لَكَ، فَلَكَ الْحَمْدُ وَلَكَ الشُّكْرُ)).  
> *(مرة واحدة)*

> [!adhkar] العافية والاستعاذة من الكفر والفقر
> ((اللَّهُمَّ عَافِنِي فِي بَدَنِي، اللَّهُمَّ عَافِنِي فِي سَمْعِي، اللَّهُمَّ عَافِنِي فِي بَصَرِي، لاَ إِلَهَ إِلاَّ أَنْتَ. اللَّهُمَّ إِنِّي أَعُوذُ بِكَ مِنَ الْكُفْرِ، وَالفَقْرِ، وَأَعُوذُ بِكَ مِنْ عَذَابِ القَبْرِ، لاَ إِلَهَ إِلاَّ أَنْتَ))
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] سؤال العفو والعافية
> ((اللَّهُمَّ إِنِّي أَسْأَلُكَ الْعَفْوَ وَالْعَافِيَةَ فِي الدُّنْيَا وَالآخِرَةِ، اللَّهُمَّ إِنِّي أَسْأَلُكَ الْعَفْوَ وَالْعَافِيَةَ: فِي دِينِي وَدُنْيَايَ وَأَهْلِي، وَمَالِي، اللَّهُمَّ اسْتُرْ عَوْرَاتِي، وَآمِنْ رَوْعَاتِي، اللَّهُمَّ احْفَظْنِي مِنْ بَينِ يَدَيَّ، وَمِنْ خَلْفِي، وَعَنْ يَمِينِي، وَعَنْ شِمَالِي، وَمِنْ فَوْقِي، وَأَعُوذُ بِعَظَمَتِكَ أَنْ أُغْتَالَ مِنْ تَحْتِي)).
> *(مرة واحدة)*

> [!adhkar] الاستعاذة من شر النفس والشيطان
> ((اللَّهُمَّ عَالِمَ الغَيْبِ وَالشَّهَادَةِ فَاطِرَ السَّمَوَاتِ وَالْأَرْضِ، رَبَّ كُلِّ شَيْءٍ وَمَلِيكَهُ، أَشْهَدُ أَنْ لاَ إِلَهَ إِلاَّ أَنْتَ، أَعُوذُ بِكَ مِنْ شَرِّ نَفْسِي، وَمِنْ شَرِّ الشَّيْطانِ وَشِركِهِ، وَأَنْ أَقْتَرِفَ عَلَى نَفْسِي سُوءاً، أَوْ أَجُرَّهُ إِلَى مُسْلِمٍ)).
> *(مرة واحدة)*

> [!adhkar] الحماية من الضر
> ((بِسْمِ اللَّهِ الَّذِي لاَ يَضُرُّ مَعَ اسْمِهِ شَيْءٌ فِي الْأَرْضِ وَلاَ فِي السّمَاءِ وَهُوَ السَّمِيعُ الْعَلِيمُ))
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] الرضا بالله رباً
> ((رَضِيتُ بِاللَّهِ رَبَّاً، وَبِالْإِسْلاَمِ دِيناً، وَبِمُحَمَّدٍ ﷺ نَبِيّاً))
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] يَا حَيُّ يَا قَيُّومُ
((يَا حَيُّ يَا قَيُّومُ بِرَحْمَتِكَ أَسْتَغِيثُ، أَصْلِحْ لِي شَأْنِي كُلَّهُ، وَلَا تَكِلْنِي إِلَى نَفْسِي طَرْفَةَ عَيْنٍ))
*(ثلاث مرات)*

> [!adhkar] دعاء الكرب والهم
((اللَّهُمَّ إِنِّي أَعُوذُ بِكَ مِنْ الْهَمِّ وَالْحَزَنِ ، وَأَعُوذُ بِكَ مِنْ الْعَجْزِ وَالْكَسَلِ ، وَأَعُوذُ بِكَ مِنْ الْجُبْنِ وَالْبُخْلِ ، وَأَعُوذُ بِكَ مِنْ غَلَبَةِ الدَّيْنِ ، وَقَهْرِ الرِّجَالِ))
*(مرة واحدة)*

> [!adhkar] الحمد لله تعالى
((يَا رَبِّ , لَكَ الْحَمْدُ كَمَا يَنْبَغِي لِجَلَالِ وَجْهِكَ , وَلِعَظِيمِ سُلْطَانِكَ))
  *(ثلاثَ مرَّاتٍ)*

> [!adhkar] الاستعاذة بكلمات الله
> ((أَعُوذُ بِكَلِمَاتِ اللَّهِ التَّامَّاتِ مِنْ شَرِّ مَا خَلَقَ))
> *(ثلاثَ مرَّاتٍ)*

> [!adhkar] الاستعاذة من الشرك والمعصية
((اللَّهُمَّ إِنَّا نَعُوذُ بِكَ مِنْ أَنْ نُشْرِكَ بِكَ شَيْئًا نَعْلَمُهُ ، وَنَسْتَغْفِرُكَ لِمَا لَا نَعْلَمُهُ))
*(مرة واحدة)*

> [!adhkar] الصلاة على النبي
> ((اللَّهُمَّ صَلِّ وَسَلِّمْ عَلَى نَبَيِّنَا مُحَمَّدٍ))
> *(عشرَ مرَّاتٍ)*

> [!adhkar] التوحيد - تاج الذكر
> ((لاَ إِلَهَ إِلاَّ اللَّهُ وَحْدَهُ لاَ شَرِيكَ لَهُ، لَهُ الْمُلْكُ وَلَهُ الْحَمْدُ، وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌ))
> *(مئة مرة)* ٫ *(عشرَ مرَّات عندَ الكَسَلِ)*

> [!adhkar] التسبيح
> ((سُبْحَانَ اللَّهِ وَبِحَمْدِهِ))
> *(مائة مرَّةٍ)*

> [!adhkar] الاستغفار
> ((أَسْتَغْفِرُ اللَّهَ وَأَتُوبُ إِلَيْهِ))
> *(مِائَةَ مَرَّةٍ فِي الْيَوْمِ)*

// دالة تحويل التكرارات إلى نصوص مشكولة بالكامل
function getRepeatText(num) {
    if (num === 1) return "*(مَرَّةً وَاحِدَةً)*";
    if (num === 2) return "*(مَرَّتَيْنِ)*";
    if (num === 3) return "*(ثَلَاثَ مَرَّاتٍ)*";
    if (num === 4) return "*(أَرْبَعَ مَرَّاتٍ)*";
    if (num === 7) return "*(سَبْعَ مَرَّاتٍ)*";
    if (num === 10) return "*(عَشْرَ مَرَّاتٍ)*";
    if (num === 100) return "*(مِائَةَ مَرَّةٍ)*";
    return `*(${num} مَرَّاتٍ)*`;
}

const storageKey = "athkar-progress";
const todayDate = new Date().toISOString().split('T')[0];

function getSavedProgress() {
    try {
        const data = localStorage.getItem(storageKey);
        if (data) {
            const parsed = JSON.parse(data);
            if (parsed.evening && parsed.evening.date === todayDate) {
                return parsed.evening.index;
            }
        }
    } catch (e) {
        console.error("Error reading progress", e);
    }
    return 0;
}

function saveProgress(index) {
    try {
        const currentData = JSON.parse(localStorage.getItem(storageKey) || "{}");
        currentData.evening = {
            date: todayDate,
            index: index
        };
        localStorage.setItem(storageKey, JSON.stringify(currentData));
    } catch (e) {
        console.error("Error saving progress", e);
    }
}

// دالة رندرة الكالوت
async function renderCallout(dhikr, container) {
    const repeatText = getRepeatText(dhikr.Repeat);
    const markdown = `> [!adhkar] ${dhikr.Title}\n> ${dhikr.Content}\n> ${repeatText}`;
    
    const tempDiv = document.createElement("div");
    tempDiv.className = "single-dhikr-wrapper";
    tempDiv.style.marginBottom = "15px";
    
    await obsidian.MarkdownRenderer.renderMarkdown(markdown, tempDiv, "", dv.component);
    container.appendChild(tempDiv);
    return tempDiv;
}
